Pasos para crear los secretos en GCP

# 1. Crear el secreto para la clave secreta de Django
gcloud secrets create django-secret-key --replication-policy="automatic"
echo -n "django-insecure-local-dev-secret-key-3n4b89v10n" | gcloud secrets versions add django-secret-key --data-file=-

# 2. Crear el secreto para la contraseña de base de datos
gcloud secrets create db-password --replication-policy="automatic"
echo -n "8mEYrnGT3eu?" | gcloud secrets versions add db-password --data-file=-

# 3. Crear el secreto para el usuario SMTP
gcloud secrets create smtp-username --replication-policy="automatic"
echo -n "9e8c40a12a4972d3cba4" | gcloud secrets versions add smtp-username --data-file=-

# 4. Crear el secreto para la contraseña SMTP
gcloud secrets create smtp-password --replication-policy="automatic"
echo -n "G73icmIZ1xHUSNeoKQDh" | gcloud secrets versions add smtp-password --data-file=-

- Otorgar permisos a la cuenta de servicio de Cloud Run 

PROJECT_NUMBER=$(gcloud projects describe $(gcloud config get-value project) --format="value(projectNumber)")

for secret in django-secret-key db-password smtp-username smtp-password; do
  gcloud secrets add-iam-policy-binding $secret \
    --member="serviceAccount:${PROJECT_NUMBER}-compute@developer.gserviceaccount.com" \
    --role="roles/secretmanager.secretAccessor"
done
